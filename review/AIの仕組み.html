  let score = 1;
  for (const e of efs){

      score = Math.max(score, (myMax - myHp) * 0.7);   // 減ったHPの7割相当の価値(満タンならほぼ0)

    }
  }
  return score;
}

function aiChooseMove(sideKey){
  const st = sides[sideKey];
  if (!st.poke || !st.moves || !st.moves.length) return 0;
  // こだわりロック中はその技しか選べない(段135)
  if (st.choiceLock && st.item && ITEM_BY_KEY[st.item] && ITEM_BY_KEY[st.item].restriction === '選択技固定'){
    const li = st.moves.findIndex(m => m && m.name === st.choiceLock.name);
    if (li >= 0) return li;
  }
  let best = -1, bestScore = -1, bestPri = -99;
  st.moves.forEach((m, i) => {
    if (!m) return;
    if (st.pp && st.pp[i] <= 0) return;   // PP切れの技は選ばない(段129)
    const score = aiScoreMove(sideKey, m);
    // ★2026-07-29 阿部さん:「先制技で倒せるなら、わざわざ後退せず先制技で倒す」
    //   確定KO(score>=1000)が複数あるときは**優先度の高い技**を選ぶ(後攻でも先に倒せる)。
    const pri = movePriority(m, st);
    if (score > bestScore || (score >= 1000 && score === bestScore && pri > bestPri)){
      bestScore = score; best = i; bestPri = pri;
    }
  });
  return best >= 0 ? best : 0;
}

function aiChooseAction(sideKey){
  const st = sides[sideKey];
  st._aiReason = null;   // ★なぜその行動を選んだか(テストと将来のフェーズ・ビューア用)
  const foe = sideKey === 'self' ? 'opp' : 'self';
  const foeSt = sides[foe];
  const best = aiChooseMove(sideKey);
  const bestScore = (st.moves && st.moves[best]) ? aiScoreMove(sideKey, st.moves[best]) : 0;
  if (bestScore >= 1000){ st.aiSwitchedLast = false; st._aiReason = 'move_確定KO'; return {type: 'move', index: best}; }   // 確定KOは最優先
  // B) 被弾予測(v3)
  if (foeSt.poke && !st.aiSwitchedLast){
    const myHp = st.currentHp == null ? realStat(st, 'hp') : st.currentHp;
    let foeMax = 0, foeMaxMove = null;
    (foeSt.moves || []).forEach(m => {
      if (!m || m.category === '変化') return;
      const r = calcDamage(foe, sideKey, m);
      if (r && !r.immune){ const avg = (r.min + r.max) / 2; if (avg > foeMax){ foeMax = avg; foeMaxMove = m; } }
    });
    const mySpd = effectiveSpeed(st), foeSpd = effectiveSpeed(foeSt);
    const slower = env.trickRoom ? (mySpd > foeSpd) : (mySpd < foeSpd);
    if (foeMaxMove && foeMax >= myHp && slower){
      // 受けられる控え: 相手の主力技タイプの被ダメ倍率が最小の子(半減以下が条件)。同率なら攻撃相性が良い子
      let swIdx = -1, swTaken = 99, swAtk = 0;
      (st.bench || []).forEach((b, i) => {
        if (!b || !b.poke || b.fainted || (b.currentHp != null && b.currentHp <= 0)) return;
        const btypes = [b.poke.type1, b.poke.type2].filter(Boolean);
        const taken = moveTypeEff(foeMaxMove.type, btypes);
        let atkEff = 0;
        (b.moves || []).forEach(m => {
          if (!m || m.category === '変化') return;
          const e = moveTypeEff(m.type, sideTypes(foeSt));
          if (e > atkEff) atkEff = e;
        });
        if (!aiSwitchAffordable(st, b)) return;   // ★設置物で死ぬ/半分以上減る控えには逃がさない
        if (taken < swTaken || (taken === swTaken && atkEff > swAtk)){ swTaken = taken; swIdx = i; swAtk = atkEff; }
      });
      if (swIdx >= 0 && swTaken <= 0.5){ st.aiSwitchedLast = true; st._aiReason = 'switch_B_受け交代'; return {type: 'switch', index: swIdx}; }
    }
  }
  // ────────────────────────────────────────────────────────────────
  // C) ★相性による「予防の交代」(2026-07-29 阿部さん「交代をうまく使うのが上級者」)
  //   考え方: 相手に弱点を突かれていて、こちらは有効打が無い → 受けられる控えへ逃がす。
  //   ただし**受けられる控えがいなければ交代しない**(逃げ場が無いなら殴る)。
  //   Bとの違い: Bは「今ターン倒される」時だけ。Cは倒されなくても**不利な組み合わせなら退く**。
  if (foeSt.poke && !st.aiSwitchedLast && bestScore < 1000){
    const myHp2 = st.currentHp == null ? realStat(st, 'hp') : st.currentHp;
    const myMax2 = realStat(st, 'hp');
    // 1) 相手の脅威: 相手の攻撃技のうち、こちらへの平均ダメージが最大のもの
    let thMove = null, thDmg = 0;
    (foeSt.moves || []).forEach(m => {
      if (!m || m.category === '変化') return;
      const r = calcDamage(foe, sideKey, m);
      if (r && !r.immune){ const avg = (r.min + r.max) / 2; if (avg > thDmg){ thDmg = avg; thMove = m; } }
    });
    // 2) こちらが有効打を持っているか(効果抜群=2倍以上の攻撃技があるなら、殴り合った方がよい)
    let myBestEff = 0;
    (st.moves || []).forEach(m => {
      if (!m || m.category === '変化') return;
      const e = moveTypeEff(m.type, sideTypes(foeSt));
      if (e > myBestEff) myBestEff = e;
    });
    // 3) 危険の条件: 相手の一撃で残HPの半分以上を持っていかれる & こちらに効果抜群が無い

    const mySpd2 = effectiveSpeed(st), foeSpd2 = effectiveSpeed(foeSt);
    const slower2 = env.trickRoom ? (mySpd2 > foeSpd2) : (mySpd2 < foeSpd2);
    const danger = thMove && thDmg >= myHp2 * 0.5 && myBestEff < 2 && slower2;
    if (danger){
      // 4) 受けられる控えを探す: 脅威技を「半減以下(特性の無効/半減も込み)」で受けられる子。
      //    同じ受けやすさなら、相手に効果抜群を返せる子を優先(受けるだけでなく攻め返せる方が良い)。
      let swIdx = -1, swTaken = 99, swAtk = 0;
      (st.bench || []).forEach((b, i) => {

        let atkEff = 0;
        (b.moves || []).forEach(m => {
          if (!m || m.category === '変化') return;
          const e = moveTypeEff(m.type, sideTypes(foeSt));
          if (e > atkEff) atkEff = e;
        });
        if (taken < swTaken || (taken === swTaken && atkEff > swAtk)){ swTaken = taken; swIdx = i; swAtk = atkEff; }
      });
      // 5) ★受けられる控えがいなければ交代しない(逃げ場が無いなら殴る)
      if (swIdx >= 0){ st.aiSwitchedLast = true; st._aiReason = 'switch_C_相性で退く'; return {type: 'switch', index: swIdx}; }
    }
  }
  if (bestScore > 5){ st.aiSwitchedLast = false; st._aiReason = 'move_有効打あり'; return {type: 'move', index: best}; }
  // A) どの技にも価値が無い(v2)
  const ftypes = foeSt.poke ? sideTypes(foeSt) : [];
  let swIdx = -1, swEff = 0;
  (st.bench || []).forEach((b, i) => {
    if (!b || !b.poke || b.fainted || (b.currentHp != null && b.currentHp <= 0)) return;
    let eff = 0;
    (b.moves || []).forEach(m => {
      if (!m || m.category === '変化') return;
      const e = moveTypeEff(m.type, ftypes);
      if (e > eff) eff = e;
    });
    if (!aiSwitchAffordable(st, b)) return;   // ★設置物で死ぬ/半分以上減る控えには逃がさない
    if (eff > swEff){ swEff = eff; swIdx = i; }
  });
  if (swIdx >= 0 && swEff > 0 && !st.aiSwitchedLast){ st.aiSwitchedLast = true; st._aiReason = 'switch_A_手詰まり'; return {type: 'switch', index: swIdx}; }
  st.aiSwitchedLast = false;
  st._aiReason = st._aiReason || 'move_通常';
  return {type: 'move', index: best};
}
